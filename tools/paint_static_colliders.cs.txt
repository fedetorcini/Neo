var scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene();if(UnityEditor.EditorApplication.isPlaying||scene.path!="Assets/Scenes/Reference/DemoLevel1_ClaroNorte.unity")throw new System.Exception("Unexpected authoring scene.");
var maps=scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<UnityEngine.Tilemaps.Tilemap>(true)).ToArray();var terrain=maps.Single(m=>m.name=="Terrain");var overlay=maps.Single(m=>m.name=="Terrain LV1");var walls=maps.Single(m=>m.name=="Colliders");var board=terrain.GetComponentInParent<NeoGridBoard>();var collider=walls.GetComponent<UnityEngine.Tilemaps.TilemapCollider2D>();var block=UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Tilemaps.Tile>("Assets/Demo/BoundaryCell.asset");if(block==null||block.colliderType!=UnityEngine.Tilemaps.Tile.ColliderType.Grid)throw new System.Exception("Expected square collision tile.");
var treeAssets=new System.Collections.Generic.HashSet<string>{"Assets/Pallete/Pallete1_486.asset","Assets/Pallete/Pallete1_487.asset"};var rockAssets=new System.Collections.Generic.HashSet<string>();foreach(int i in new[]{107,114,115,116,117,121,122,124,125,145,211,380})rockAssets.Add("Assets/Pallete/Pallete1_"+i+".asset");foreach(int i in new[]{1,5,7,16})rockAssets.Add("Assets/Pallete/Tilesets/RockExtras_"+i+".asset");
var treeCells=new System.Collections.Generic.HashSet<UnityEngine.Vector3Int>();var stoneCells=new System.Collections.Generic.HashSet<UnityEngine.Vector3Int>();foreach(var map in new[]{terrain,overlay})foreach(var c in map.cellBounds.allPositionsWithin){var tile=map.GetTile(c);if(tile==null)continue;var asset=UnityEditor.AssetDatabase.GetAssetPath(tile);var target=walls.WorldToCell(map.GetCellCenterWorld(c));if(treeAssets.Contains(asset))treeCells.Add(target);if(rockAssets.Contains(asset))stoneCells.Add(target);}
var obstacles=new System.Collections.Generic.HashSet<UnityEngine.Vector3Int>(treeCells);obstacles.UnionWith(stoneCells);foreach(var c in obstacles){var tile=terrain.GetTile(terrain.WorldToCell(walls.GetCellCenterWorld(c)));if(tile!=null&&(tile.name=="PathTile"||tile.name=="Pallete1_207"))throw new System.Exception("Would block a path or staircase: "+c);}
UnityEditor.Undo.RecordObjects(new UnityEngine.Object[]{walls,board,collider,walls.gameObject},"Pintar bases y piedras en Colliders");foreach(var c in obstacles)walls.SetTile(c,block);walls.gameObject.layer=UnityEngine.LayerMask.NameToLayer("Collider");collider.enabled=true;collider.isTrigger=false;board.walls=walls;var b=terrain.cellBounds;var old=board.navigationBounds;board.navigationBounds=new UnityEngine.BoundsInt(UnityEngine.Vector3Int.Min(old.min,b.min),UnityEngine.Vector3Int.Max(old.max,b.max)-UnityEngine.Vector3Int.Min(old.min,b.min));
walls.RefreshAllTiles();walls.CompressBounds();collider.ProcessTilemapChanges();UnityEngine.Physics2D.SyncTransforms();
int testedBlocked=0;foreach(var c in obstacles){var center=walls.GetCellCenterWorld(c);if(!collider.OverlapPoint(center)||!board.StaticBlocked(new UnityEngine.Vector2Int(c.x,c.y)))throw new System.Exception("Missing physical or movement block: "+c);testedBlocked++;}
int openPath=0;foreach(var c in terrain.cellBounds.allPositionsWithin){var tile=terrain.GetTile(c);if(tile==null||(tile.name!="PathTile"&&tile.name!="Pallete1_207"))continue;var center=terrain.GetCellCenterWorld(c);if(collider.OverlapPoint(center)||board.StaticBlocked(new UnityEngine.Vector2Int(c.x,c.y)))throw new System.Exception("Path blocked: "+c);openPath++;}
var actorCells=board.actors.Where(a=>a!=null).Select(a=>NeoGridBoard.Cell(a.transform.position)).ToArray();if(actorCells.Any(c=>board.StaticBlocked(c)))throw new System.Exception("An actor starts in a static obstacle.");
UnityEditor.EditorUtility.SetDirty(board);UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene);
var report=new{scene=scene.path,treeBaseCells=treeCells.Count,stoneCells=stoneCells.Count,uniqueBlocks=obstacles.Count,physicalCentersChecked=testedBlocked,walkablePathAndStairCellsChecked=openPath,navigationBounds=board.navigationBounds.ToString(),layer=UnityEngine.LayerMask.LayerToName(walls.gameObject.layer),isTrigger=collider.isTrigger};return report;
